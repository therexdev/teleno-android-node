cmake_minimum_required(VERSION 3.19)
project(teleno_android_bridge LANGUAGES C)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The Android bridge requires the arm64-v8a NDK toolchain")
endif()
find_package(OpenSSL REQUIRED)
add_library(teleno_keys SHARED keys.c files.c)
target_link_libraries(teleno_keys PRIVATE OpenSSL::Crypto)
target_compile_options(teleno_keys PRIVATE -Wall -Wextra -Werror)
add_executable(teleno_runner runner.c)
target_compile_options(teleno_runner PRIVATE -Wall -Wextra -Werror)
target_link_options(teleno_keys PRIVATE -Wl,-z,max-page-size=16384)
target_link_options(teleno_runner PRIVATE -Wl,-z,max-page-size=16384)
